add_executable(dmjson_native ../../src/dmjson.c ../dmjson_test.c sal.c main.c)
add_executable(dmjson_roundtrip ../../src/dmjson.c sal.c roundtrip.c)
option(DMJSON_SANITIZERS "Enable address/undefined behavior sanitizers" ON)
foreach(target dmjson_native dmjson_roundtrip)
    target_include_directories(${target} PRIVATE . ../../include)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic -Wvla -Werror=vla)
    if(DMJSON_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
add_test(NAME dmjson_native COMMAND dmjson_native)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME dmjson_differential COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/differential.py $<TARGET_FILE:dmjson_roundtrip>)
endif()
